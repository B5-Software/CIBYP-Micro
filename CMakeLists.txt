cmake_minimum_required(VERSION 3.16)
project(cibyp_micro VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(cibyp_micro src/json.c src/micro.c src/fragments.c)
target_include_directories(cibyp_micro PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cibyp_micro PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(micro-simulator examples/simulator.c)
target_link_libraries(micro-simulator PRIVATE cibyp_micro)
enable_testing()
add_executable(micro-tests tests/test_micro.c)
target_link_libraries(micro-tests PRIVATE cibyp_micro)
add_test(NAME micro-contract COMMAND micro-tests)
